`Checkbox` is an accessible checkbox built on Radix UI's `@radix-ui/react-checkbox` primitive. A check icon appears when the box is checked.


## Interactive example

<Specimen label="Checkbox · Live example">
  <CheckboxPreview />

```tsx
import { useState } from "react";
import * as UX from "@clidey/ux";
import { Demo, Status } from "./demo-shell";

export function CheckboxPreview() {
  const [items, setItems] = useState(["Design"]);
  const options = ["Design", "Engineering", "Product"];
  return (
    <Demo hint="Choose individual teams or use the indeterminate parent to select all.">
      <div className="ux-demo-form">
        <div className="ux-demo-row">
          <UX.Checkbox
            id="check-all"
            checked={items.length === 3 ? true : items.length ? "indeterminate" : false}
            onCheckedChange={(v) => setItems(v ? options : [])}
          />
          <UX.Label htmlFor="check-all">All teams</UX.Label>
        </div>
        <UX.Separator />
        {options.map((team) => (
          <div className="ux-demo-row" key={team}>
            <UX.Checkbox
              id={"check-" + team}
              checked={items.includes(team)}
              onCheckedChange={(v) =>
                setItems(v ? [...items, team] : items.filter((x) => x !== team))
              }
            />
            <UX.Label htmlFor={"check-" + team}>{team}</UX.Label>
          </div>
        ))}
        <Status>{items.length} teams selected</Status>
      </div>
    </Demo>
  );
}
```
</Specimen>

The code uses shared demo presentation helpers. Their complete source and styling are included in [the LLM build context](/llms.txt).


## Import

```tsx
import { Checkbox } from '@clidey/ux';
```

## Basic usage

```tsx
<Checkbox id="terms" />
```

## With label

Always pair a checkbox with a visible `<Label>` for accessibility:

```tsx
import { Checkbox, Label } from '@clidey/ux';

<div className="flex items-center gap-2">
  <Checkbox id="terms" />
  <Label htmlFor="terms">
    I agree to the terms and conditions
  </Label>
</div>
```

## Controlled

```tsx
import { useState } from 'react';

function ControlledCheckbox() {
  const [checked, setChecked] = useState(false);

  return (
    <div className="flex items-center gap-2">
      <Checkbox
        id="notify"
        checked={checked}
        onCheckedChange={setChecked}
      />
      <Label htmlFor="notify">Email notifications</Label>
    </div>
  );
}
```

## Disabled

```tsx
<div className="flex items-center gap-2">

```

## Checkbox list

```tsx
const options = [
  { id: 'email', label: 'Email notifications' },
  { id: 'sms', label: 'SMS notifications' },
  { id: 'push', label: 'Push notifications' },
];

function NotificationPrefs() {
  const [selected, setSelected] = useState<string[]>([]);

  function toggle(id: string) {
    setSelected((prev) =>
      prev.includes(id) ? prev.filter((x) => x !== id) : [...prev, id]
    );
  }

  return (
    <div className="space-y-2">
      {options.map((opt) => (
        <div key={opt.id} className="flex items-center gap-2">
          <Checkbox
            id={opt.id}
            checked={selected.includes(opt.id)}
            onCheckedChange={() => toggle(opt.id)}
          />
          <Label htmlFor={opt.id}>{opt.label}</Label>
        </div>
      ))}
    </div>
  );
}
```

## Props

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `checked` | `boolean \| "indeterminate"` | — | Controlled checked state. |
| `defaultChecked` | `boolean` | `false` | Uncontrolled default state. |
| `onCheckedChange` | `(checked: boolean \| "indeterminate") => void` | — | Called when the state changes. |
| `disabled` | `boolean` | `false` | Disables interaction. |
| `required` | `boolean` | `false` | Marks the checkbox as required in a form. |
| `name` | `string` | — | Form field name for native form submission. |
| `value` | `string` | `"on"` | Value submitted with form data when checked. |
| `id` | `string` | — | Links to a `<Label htmlFor>`. |
| `className` | `string` | — | Additional CSS classes. |
